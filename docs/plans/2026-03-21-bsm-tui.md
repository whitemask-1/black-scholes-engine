# BSM TUI Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a reactive Textual TUI in `bsm_tui.py` that lets users explore live BSM options chains by entering a ticker, selecting an expiration date, and choosing call/put.

**Architecture:** Single `App` class with an inline header bar (ticker `Input`, expiration `Select`, type `Select`) and a full-width `DataTable` below. Ticker validation runs on a 500ms debounce using `set_timer`; all network/compute work runs in Textual background workers via `@work(thread=True)` to avoid blocking the event loop.

**Tech Stack:** Textual 8.1.1, yfinance 1.2.0, pandas — existing `market.analyze_live()` is reused unchanged.

```python
import yfinance as yf
from textual.app import App, ComposeResult
from textual.containers import Horizontal, Vertical
from textual.widgets import DataTable, Footer, Input, Label, Select
from textual import work

from market import analyze_live


class BSMApp(App):
    CSS = """
    #header {
        height: 3;
        padding: 0 1;
        background: $panel;
        align: left middle;
    }
    #header Label {
        margin: 0 1 0 0;
        color: $text-muted;
    }
    #ticker {
        width: 12;
    }
    #expiration {
        width: 20;
    }
    #option-type {
        width: 10;
    }
    #chain {
        height: 1fr;
    }
    #status {
        height: 1;
        padding: 0 1;
        background: $panel;
        color: $text-muted;
    }
    """

    def compose(self) -> ComposeResult:
        with Horizontal(id="header"):
            yield Label("Ticker:")
            yield Input(placeholder="AAPL", id="ticker")
            yield Label("Expiration:")
            yield Select([], id="expiration", disabled=True, prompt="—")
            yield Label("Type:")
            yield Select(
                [("call", "call"), ("put", "put")],
                id="option-type",
                value="call",
            )
        yield DataTable(id="chain")
        yield Label("", id="status")


if __name__ == "__main__":
    BSMApp().run()
```

### Task 2: Ticker debounce + expiration loading

**Files:**
- Modify: `bsm_tui.py`

When the user types in the ticker box, wait 500ms then fetch available expiration dates. If valid dates come back, populate and enable the expiration `Select`. If not, disable it and clear the table.

**Step 1: Add the debounced input handler and worker**

Add these methods to `BSMApp` (before the closing `if __name__` block):

```python
    def on_input_changed(self, event: Input.Changed) -> None:
        if event.input.id != "ticker":
            return
        try:
            self._ticker_timer.stop()
        except AttributeError:
            pass
        self._ticker_timer = self.set_timer(0.5, self._fetch_expirations)

    @work(thread=True)
    def _fetch_expirations(self) -> None:
        symbol = self.query_one("#ticker", Input).value.strip().upper()
        if not symbol:
            self.call_from_thread(self._clear_all)
            return
        try:
            expirations = yf.Ticker(symbol).options
        except Exception:
            self.call_from_thread(self._clear_all)
            return
        if expirations:
            self.call_from_thread(self._set_expirations, symbol, expirations)
        else:
            self.call_from_thread(self._clear_all)

    def _set_expirations(self, symbol: str, expirations: tuple) -> None:
        select = self.query_one("#expiration", Select)
        select.set_options([(e, e) for e in expirations])
        select.disabled = False
        self._set_status(f"{symbol}: {len(expirations)} expiration dates available")

    def _clear_all(self) -> None:
        select = self.query_one("#expiration", Select)
        select.set_options([])
        select.disabled = True
        self.query_one("#chain", DataTable).clear(columns=True)
        self._set_status("")

    def _set_status(self, msg: str) -> None:
        self.query_one("#status", Label).update(msg)
```

### Task 3: Chain loading on select change

**Files:**
- Modify: `bsm_tui.py`

When the user changes the expiration date or option type, fetch and display the options chain.

**Step 1: Add the select change handler and chain worker**

Add these methods to `BSMApp`:

```python
    def on_select_changed(self, event: Select.Changed) -> None:
        if event.select.id in ("expiration", "option-type"):
            expiration = self.query_one("#expiration", Select).value
            if expiration is Select.BLANK or expiration is None:
                return
            self._fetch_chain()

    @work(thread=True)
    def _fetch_chain(self) -> None:
        symbol = self.query_one("#ticker", Input).value.strip().upper()
        expiration = self.query_one("#expiration", Select).value
        option_type = self.query_one("#option-type", Select).value

        if not symbol or expiration is Select.BLANK or expiration is None:
            return

        self.call_from_thread(
            self._set_status,
            f"Loading {symbol} {option_type} chain for {expiration}...",
        )
        try:
            df = analyze_live(symbol, expiration, option_type)
            self.call_from_thread(self._populate_table, df)
        except Exception as e:
            self.call_from_thread(self._set_status, f"Error: {e}")

    def _populate_table(self, df) -> None:
        table = self.query_one("#chain", DataTable)
        table.clear(columns=True)
        for col in df.columns:
            table.add_column(col.upper(), key=col)
        for _, row in df.iterrows():
            table.add_row(
                *[f"{v:.4f}" if isinstance(v, float) else str(v) for v in row]
            )
        self._set_status(
            f"{len(df)} rows loaded"
        )
```
